function buildReport(start,end){ /* ===================================================== تحديد الأسبوع السابق ===================================================== */ const previousWeek = getWeekFromDate( new Date( new Date(start+"T00:00:00").getTime() - 24*60*60*1000 ) .toISOString() .split("T")[0] ); /* ===================================================== التحقق هل توجد بيانات فعلية في الأسبوع السابق ===================================================== */ const previousWeekHasData = allData.some(item => item.date >= previousWeek[0] && item.date <= previousWeek[1] ); let results=[]; /* ===================================================== حساب الفروع ===================================================== */ branches.forEach(branch=>{ const currentAverage = weeklyAverage( branch, start, end ); const previousAverage = previousWeekHasData ? weeklyAverage( branch, previousWeek[0], previousWeek[1] ) : null; if(currentAverage !== null){ let change=null; /* إذا لا توجد بيانات للأسبوع السابق لا نحسب التغيير */ if( previousWeekHasData && previousAverage !== null ){ change = currentAverage - previousAverage; } results.push({ branch:branch, current:currentAverage, previous:previousAverage, change:change }); } }); /* ===================================================== ترتيب الفروع ===================================================== */ results.sort( (a,b)=> b.current-a.current ); /* ===================================================== بداية التقرير ===================================================== */ let html=`

📊 التقرير الأسبوعي الرئيسي

الفترة: ${start} إلى ${end}
الأسبوع السابق: ${ previousWeekHasData ? previousWeek[0]+" إلى "+previousWeek[1] : "لا توجد بيانات للأسبوع السابق" }

🏆 الترتيب الرئيسي للفروع

`; if(!results.length){ html+=` `; }else{ results.forEach( (item,index)=>{ const percentages = weeklySectionPercentages( item.branch, start, end ); let changeText="—"; let changeClass="same"; if(item.change!==null){ if(item.change>0){ changeText = "+"+ item.change.toFixed(1)+ " نقطة"; changeClass="good"; } else if(item.change<0){ changeText = item.change.toFixed(1)+ " نقطة"; changeClass="bad"; } else{ changeText = "0.0 نقطة"; } } const status = getStatus(item.current); html+=` ${percentages.map(p=>` `).join("")} `; }); } html+=`
الترتيب الفرع الأمامية المطبخ السلامة التشغيل الكاشير المخزن الممنوعة السكن الاستجابة نسبة الأسبوع الأسبوع السابق التغيير الحالة
لا توجد تقييمات لهذا الأسبوع
${index+1} ${item.branch} ${ p===null ?"—" :p.toFixed(1)+"%" } ${item.current.toFixed(1)}% ${ item.previous===null ?"—" :item.previous.toFixed(1)+"%" } ${changeText} ${status.text}

📈📉 حركة الفروع مقارنة بالأسبوع السابق

`; /* إذا لا توجد بيانات للأسبوع السابق نعرض رسالة واضحة بدل "لا توجد فروع متحسنة" */ if(!previousWeekHasData){ html+=`

ℹ️ لا توجد مقارنة أسبوعية

لا توجد تقييمات محفوظة للأسبوع السابق، لذلك لن يتم احتساب التحسن أو التراجع في هذا التقرير.

`; }else{ html+=`
`; const improved = results .filter(item=> item.change!==null && item.change>0.05 ) .sort( (a,b)=> b.change-a.change ); const declined = results .filter(item=> item.change!==null && item.change<-0.05 ) .sort( (a,b)=> a.change-b.change ); const unchanged = results .filter(item=> item.change!==null && Math.abs(item.change)<=0.05 ); html+=`

📈 الفروع المتحسنة

`; if(!improved.length){ html+=`

لا توجد فروع متحسنة.

`; }else{ improved.forEach(item=>{ html+=`
${item.branch} +${item.change.toFixed(1)} نقطة
`; }); } html+=`

📉 الفروع المتراجعة

`; if(!declined.length){ html+=`

لا توجد فروع متراجعة.

`; }else{ declined.forEach(item=>{ html+=`
${item.branch} ${item.change.toFixed(1)} نقطة
`; }); } html+=`

➖ الفروع الثابتة

`; if(!unchanged.length){ html+=`

لا توجد فروع ثابتة.

`; }else{ unchanged.forEach(item=>{ html+=`
${item.branch} 0.0 نقطة
`; }); } html+=`
`; } /* ===================================================== نقاط القوة والضعف لكل فرع ===================================================== */ html+=`

📋 نقاط القوة والضعف لكل فرع

`; results.forEach(item=>{ const percentages = weeklySectionPercentages( item.branch, start, end ); const valid=[]; sections.forEach( (section,index)=>{ if(percentages[index]!==null){ valid.push({ name:section.name, percentage: percentages[index] }); } } ); const weakest = [...valid] .sort( (a,b)=> a.percentage-b.percentage ) .slice(0,3); const strongest = [...valid] .sort( (a,b)=> b.percentage-a.percentage ) .slice(0,3); const status = getStatus(item.current); html+=`

🏪 ${item.branch}

النسبة الكلية: ${item.current.toFixed(1)}%   ${status.text}
`; valid.forEach(section=>{ let className=""; if(section.percentage<70){ className="section-weak"; } else if(section.percentage>=90){ className="section-strong"; } html+=` `; }); html+=`
القسم نسبة الأسبوع
${section.name} ${section.percentage.toFixed(1)}%

🔴 أضعف 3 أقسام

`; weakest.forEach(section=>{ html+=`
${section.name} — ${section.percentage.toFixed(1)}%
`; }); html+=`

🟢 أقوى 3 أقسام

`; strongest.forEach(section=>{ html+=`
${section.name} — ${section.percentage.toFixed(1)}%
`; }); html+=`
`; }); /* ===================================================== أكثر نقاط الضعف تكراراً ===================================================== */ const weaknessCounts = sections.map(section=>({ name:section.name, count:0, total:0, sum:0 })); results.forEach(branch=>{ const percentages = weeklySectionPercentages( branch.branch, start, end ); percentages.forEach( (percentage,index)=>{ if(percentage===null){ return; } weaknessCounts[index].total++; weaknessCounts[index].sum += percentage; if(percentage<70){ weaknessCounts[index].count++; } } ); }); const repeatedWeaknesses = weaknessCounts .filter(item=> item.count>0 ) .sort( (a,b)=> b.count-a.count ); html+=`

🔴 أكثر نقاط الضعف تكراراً

`; if(!repeatedWeaknesses.length){ html+=`

لا توجد أقسام أقل من 70% هذا الأسبوع.

`; }else{ repeatedWeaknesses.forEach(item=>{ html+=`
${item.name} ضعيف في ${item.count} فرع
`; }); } html+=`
`; /* ===================================================== متوسط كل قسم بين الفروع ===================================================== */ const sectionOverall = sections.map( (section,index)=>{ let sum=0; let count=0; results.forEach(branch=>{ const percentages = weeklySectionPercentages( branch.branch, start, end ); if(percentages[index]!==null){ sum += percentages[index]; count++; } }); return{ name:section.name, average: count ?sum/count :null }; } ); html+=`

📊 متوسط كل قسم بين الفروع

`; sectionOverall.forEach(item=>{ html+=` `; }); html+=`
القسم متوسط النسبة
${item.name} ${ item.average===null ?"—" :item.average.toFixed(1)+"%" }
`; return html; }
